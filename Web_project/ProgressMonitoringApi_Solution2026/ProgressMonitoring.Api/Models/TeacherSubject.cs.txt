namespace ProgressMonitoring.Api.Models;

// "This teacher teaches this subject to this grade."
// Chosen by the teacher when they onboard, and the only thing that decides
// which learners, assessments and marks they can see.
public class TeacherSubject
{
    public Guid Id { get; set; }
    public Guid TeacherId { get; set; }
    public Guid SubjectId { get; set; }
    public string Grade { get; set; } = string.Empty;
    public int AcademicYear { get; set; }
}
